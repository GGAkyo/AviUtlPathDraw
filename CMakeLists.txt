cmake_minimum_required(VERSION 3.24)
project(AviUtlPathDraw LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "AviUtl1 requires a 32-bit Windows build; use -A Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BLEND2D_STATIC TRUE CACHE BOOL "" FORCE)
set(BLEND2D_TEST FALSE CACHE BOOL "" FORCE)
set(ASMJIT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/asmjit" CACHE PATH "" FORCE)
add_subdirectory(vendor/blend2d EXCLUDE_FROM_ALL)
add_library(path_draw SHARED src/path_draw.cpp src/path_draw.def src/path_draw.rc)
target_compile_features(path_draw PRIVATE cxx_std_17)
target_include_directories(path_draw PRIVATE sdk)
target_compile_options(path_draw PRIVATE /utf-8)
target_link_libraries(path_draw PRIVATE blend2d::blend2d user32 gdi32 comdlg32)
set_target_properties(path_draw PROPERTIES PREFIX "" SUFFIX ".auf")
